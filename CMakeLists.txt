cmake_minimum_required(VERSION 3.28)
project(TwitchDockThemes VERSION 0.5.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.11.1 EXACT REQUIRED COMPONENTS Core Gui Widgets)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/bridge.js")
if(NOT BROWSER_HEADER_DIR)
  set(BROWSER_HEADER_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor")
endif()
if(NOT OBS_SOURCE_DIR OR NOT OBS_IMPORT_DIR OR NOT BROWSER_HEADER_DIR)
  message(FATAL_ERROR "Set OBS_SOURCE_DIR, OBS_IMPORT_DIR, BROWSER_HEADER_DIR. See README.")
endif()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated")
file(WRITE "${CMAKE_BINARY_DIR}/generated/obsconfig.h" "#pragma once\n#define OBS_RELEASE_CANDIDATE 0\n#define OBS_BETA 0\n")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/bridge.js" BRIDGE)
file(WRITE "${CMAKE_BINARY_DIR}/generated/bridge-generated.hpp" "#pragma once\ninline constexpr char bridgeScript[] = R\"BRIDGE(${BRIDGE})BRIDGE\";\n")
add_library(obs_import INTERFACE)
target_include_directories(obs_import INTERFACE
  "${OBS_SOURCE_DIR}/libobs" "${OBS_SOURCE_DIR}/frontend/api"
  "${BROWSER_HEADER_DIR}" "${CMAKE_BINARY_DIR}/generated")
add_library(twitch-dock-themes MODULE src/plugin.cpp src/controller.cpp)
target_link_libraries(twitch-dock-themes PRIVATE obs_import Qt6::Widgets
  "${OBS_IMPORT_DIR}/obs.lib" "${OBS_IMPORT_DIR}/obs-frontend-api.lib")
target_compile_options(twitch-dock-themes PRIVATE /W4 /utf-8)
set_target_properties(twitch-dock-themes PROPERTIES PREFIX "")
configure_file(src/version.rc.in "${CMAKE_BINARY_DIR}/generated/version.rc" @ONLY)
target_sources(twitch-dock-themes PRIVATE "${CMAKE_BINARY_DIR}/generated/version.rc")
add_executable(dock-theme-tests tests/harness.cpp src/controller.cpp "${BROWSER_HEADER_DIR}/browser-panel.hpp")
set_target_properties(dock-theme-tests PROPERTIES AUTOMOC ON)
target_link_libraries(dock-theme-tests PRIVATE obs_import Qt6::Widgets)
target_include_directories(dock-theme-tests PRIVATE src)
target_compile_definitions(dock-theme-tests PRIVATE PALETTE_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/tests/palettes.json")
target_compile_options(dock-theme-tests PRIVATE /W4 /utf-8)
enable_testing()
add_test(NAME dock-controller COMMAND dock-theme-tests)
set_tests_properties(dock-controller PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=minimal")

add_executable(settings-ui-tests tests/settings-ui.cpp src/controller.cpp)
target_link_libraries(settings-ui-tests PRIVATE obs_import Qt6::Widgets
  "${OBS_IMPORT_DIR}/obs.lib" "${OBS_IMPORT_DIR}/obs-frontend-api.lib")
target_compile_options(settings-ui-tests PRIVATE /W4 /utf-8)
add_test(NAME settings-ui COMMAND settings-ui-tests)
set_tests_properties(settings-ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=minimal" TIMEOUT 20)
